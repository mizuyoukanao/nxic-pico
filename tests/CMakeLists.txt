cmake_minimum_required(VERSION 3.13)
project(nxic_host_tests C)
set(CMAKE_C_STANDARD 11)
if (NOT PICO_SDK_PATH)
    message(FATAL_ERROR "Pass -DPICO_SDK_PATH=... (same SDK used for firmware)")
endif()
set(NXIC_SRC ${CMAKE_CURRENT_LIST_DIR}/../src)
set(LWIP ${PICO_SDK_PATH}/lib/lwip/src)
set(TINYUSB ${PICO_SDK_PATH}/lib/tinyusb/src)
add_compile_options(-Wall -Wextra -Werror -ffunction-sections -fdata-sections)
add_compile_options(-Wno-deprecated-declarations)
add_compile_definitions(CFG_TUSB_OS=OPT_OS_NONE CFG_TUSB_MCU=OPT_MCU_NONE TUP_DCD_ENDPOINT_MAX=8)
include_directories(stubs ${NXIC_SRC} ${LWIP}/include ${TINYUSB})
file(GLOB LWIP_CORE ${LWIP}/core/*.c ${LWIP}/core/ipv4/*.c)
add_executable(network test_network.c ${NXIC_SRC}/dhcp_server.c ${LWIP_CORE} ${LWIP}/netif/ethernet.c)
target_link_options(network PRIVATE -Wl,--gc-sections)
enable_testing()
add_test(NAME network COMMAND network)
# A control build using the original lwIP setting must reproduce the drop.
file(READ ${NXIC_SRC}/lwipopts.h ORIGINAL_OPTS)
string(REPLACE "#define LWIP_IP_ACCEPT_UDP_PORT(port) ((port) == PP_HTONS(67))" "" ORIGINAL_OPTS "${ORIGINAL_OPTS}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/before/lwipopts.h "${ORIGINAL_OPTS}")
add_executable(network_before test_network.c ${NXIC_SRC}/dhcp_server.c ${LWIP_CORE} ${LWIP}/netif/ethernet.c)
target_include_directories(network_before BEFORE PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/before)
target_compile_definitions(network_before PRIVATE TEST_DHCP_BEFORE_FIX)
target_link_options(network_before PRIVATE -Wl,--gc-sections)
add_test(NAME network_before COMMAND network_before)
add_executable(host test_host.c ${NXIC_SRC}/hid_input.c ${NXIC_SRC}/gamepad.c ${NXIC_SRC}/mapping.c ${NXIC_SRC}/switch2_host.c)
add_test(NAME host COMMAND host)

# Use exactly the NCM backport that is compiled into the firmware, alongside
# an unpatched control build that reproduces the rejected USB parameters.
include(../cmake/ncm_windows_compat.cmake)
nxic_prepare_ncm_driver("${PICO_SDK_PATH}/lib/tinyusb" NCM_COMPAT_SOURCE)
foreach(variant IN ITEMS ncm ncm_before)
    if(variant STREQUAL "ncm")
        set(ncm_driver "${NCM_COMPAT_SOURCE}")
    else()
        set(ncm_driver "${TINYUSB}/class/net/ncm_device.c")
    endif()
    add_executable(${variant} test_ncm.c ${NXIC_SRC}/usb_descriptors.c ${ncm_driver})
    target_include_directories(${variant} PRIVATE ${TINYUSB}/class/net)
    add_test(NAME ${variant} COMMAND ${variant})
endforeach()
target_compile_definitions(ncm_before PRIVATE TEST_NCM_BEFORE_FIX)
